  async function addFiles(fileList) {
    const incoming = Array.from(fileList || []).filter(f => /^image\//.test(f.type) || /\.(jpe?g|png|webp|heic|heif)$/i.test(f.name));
    if (!incoming.length) return;

    resetResult();
    for (const file of incoming) {
      const duplicate = state.items.some(x => x.file.name === file.name && x.file.size === file.size && x.file.lastModified === file.lastModified);
      if (duplicate) continue;
      const item = {
        file,
        date: null,
        state: 'reading',
        message: '',
        thumbUrl: isHeic(file) ? null : URL.createObjectURL(file)
      };
      state.items.push(item);
      renderList();
      try {
        item.date = await readCaptureDate(file);
        if (item.date) {
          item.state = 'ready';
          item.message = '';
        } else {
          item.state = 'skipped';
          item.message = '找不到 EXIF 實際拍攝日期';
        }
      } catch (err) {
        item.state = 'error';
        item.message = err?.message || 'EXIF 讀取失敗';
      }
      renderList();
    }
    updateControls();
  }

  function updateControls() {
    const ready = state.items.filter(x => x.state === 'ready' || x.state === 'done').length;
    fileCount.textContent = state.items.length ? `已選取 ${state.items.length} 張・可處理 ${ready} 張` : '尚未選取照片';
    clearBtn.disabled = !state.items.length || state.processing;
    processBtn.disabled = !ready || state.processing;
  }

  function resetResult() {
    state.lastZipBlob = null;
    state.lastZipName = '';
    resultArea.classList.add('hidden');
    saveZipBtn.classList.add('hidden');
  }

  function clearAll() {
    for (const item of state.items) if (item.thumbUrl) URL.revokeObjectURL(item.thumbUrl);
    state.items = [];
    fileInput.value = '';
    resetResult();
    progressArea.classList.add('hidden');
    renderList();
    updateControls();
  }

  async function decodeToBitmap(file) {
    let sourceBlob = file;
    if (isHeic(file)) {
      if (!window.heic2any) throw new Error('HEIC 模組尚未載入，請確認網路後重新開啟 App。');
      let converted = await heic2any({blob: file, toType: 'image/jpeg', quality: 1});
      if (Array.isArray(converted)) converted = converted[0];
      sourceBlob = converted;
    }

    if ('createImageBitmap' in window) {
      try {
        return await createImageBitmap(sourceBlob, {imageOrientation: 'from-image'});
      } catch (_) {
        return await createImageBitmap(sourceBlob);
      }
    }

    return await new Promise((resolve, reject) => {
      const url = URL.createObjectURL(sourceBlob);
      const img = new Image();
      img.onload = () => { URL.revokeObjectURL(url); resolve(img); };
      img.onerror = () => { URL.revokeObjectURL(url); reject(new Error('照片解碼失敗')); };
      img.src = url;
    });
  }

  function stampScale() {
    if (stampSize.value === 'small') return 0.031;
    if (stampSize.value === 'large') return 0.048;
    return 0.039;
  }

  function drawPoly(ctx, pts) {
    ctx.beginPath();
    ctx.moveTo(pts[0][0], pts[0][1]);
    for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i][0], pts[i][1]);
    ctx.closePath();
    ctx.fill();
  }
